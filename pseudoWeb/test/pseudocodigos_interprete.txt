CASO 1: ARITMETICA BASICA
====================
Accion aritmetica_basica : ES
Ambiente
  a, b, total : Entero
Proceso
  a := 4
  b := 6
  total := a + b * 2
  Escribir(total)
FinAccion

============================
CASO 2: CONDICIONAL SI / SINO
============================
Accion condicional_simple : ES
Ambiente
  edad : Entero
Proceso
  edad := 17
  Si edad >= 18 Entonces
    Escribir("mayor de edad")
  SiNo
    Escribir("menor de edad")
  FinSi
FinAccion

=========================
CASO 3: BUCLE PARA (HASTA)
=========================
Accion para_hasta : ES
Ambiente
  i, suma : Entero
Proceso
  suma := 0
  Para i := 1 Hasta 5 Hacer
    suma := suma + i
  FinPara
  Escribir(suma)
FinAccion

=====================================
CASO 4: BUCLE PARA CON PASO NEGATIVO
=====================================
Accion para_paso_negativo : ES
Ambiente
  i, cuenta : Entero
Proceso
  cuenta := 0
  Para i := 5 Hasta 1, -1 Hacer
    cuenta := cuenta + i
  FinPara
  Escribir(cuenta)
FinAccion

========================
CASO 5: BUCLE MIENTRAS
========================
Accion mientras_controlado : ES
Ambiente
  n : Entero
Proceso
  n := 1
  Mientras n <= 3 Hacer
    Escribir(n)
    n := n + 1
  FinMientras
FinAccion

=============================
CASO 6: REPETIR HASTAQUE
=============================
Accion repetir_hastaque : ES
Ambiente
  contador : Entero
Proceso
  contador := 0
  Repetir
    contador := contador + 1
    Escribir(contador)
  HastaQue contador >= 3
FinAccion

===============================
CASO 7: SEGUN CON CASO POR DEFECTO
===============================
Accion segun_opcion : ES
Ambiente
  opcion : Entero
Proceso
  opcion := 2
  Segun opcion Hacer
    1: Escribir("uno")

    Otro: Escribir("otro")
  FinSegun
FinAccion

CASO 10: COMBINADO (IF + PARA)
=================================
Accion combinado_if_para : ES
Ambiente
  i, pares : Entero
Proceso
  pares := 0
  Para i := 1 Hasta 10 Hacer
    Si i MOD 2 = 0 Entonces
      pares := pares + 1
    FinSi
  FinPara
  Escribir(pares)
FinAccion
 
Accion ordenarNumeros : ES
Ambiente
  a, b, c, temp : Entero
Proceso
  Escribir("Ingrese tres numeros:")
  Leer(a, b, c)
  Si a > b Entonces
    temp := a
    a := b
    b := temp
  FinSi
  Si a > c Entonces
    temp := a
    a := c
    c := temp
  FinSi
  Si b > c Entonces
    temp := b
    b := c
    c := temp
  FinSi
  Escribir(a, b, c)
FinAccion

Accion funciones_complejas : ES
  Ambiente
    resultado : Entero

  Funcion operaciones(n : Entero) : Entero
  Ambiente
    a, b, c, d : Entero
  Proceso
    // Inicializamos las 4 variables con operaciones basadas en n
    a := n + 2
    b := n * 3
    c := n - 1
    d := n ^ 2   // cuadrado de n

    // Combinamos las operaciones
    resultado := (a + b) - (c + d)

    operaciones := resultado
  FinFuncion


Accion condicional_compuesto : ES
  Ambiente
    edad: Entero
    altura: Real
    extranjero: Logico
    inputExtranjero: Caracter
  Proceso
    Escribir("Ingrese su edad y altura:")
    Leer(edad)
    Si edad < 18 Entonces
      Escribir("Es menor")
    FinSi
    Leer(altura)
    Si altura < 1.70 Entonces
      Escribir("Es bajo")
    FinSi
    Escribir("¿Es extranjero? (Verdadero/Falso):")
    Leer(inputExtranjero)
    Escribir(inputExtranjero)
    Si inputExtranjero = 'V' o inputExtranjero = 'v' Entonces
      extranjero := Verdadero
    SiNo
      extranjero := Falso
    FinSi
    Si (edad >= 18 y altura >= 1.70) o (extranjero = Verdadero) Entonces
      Si extranjero = Verdadero Entonces
        Si altura < 1.70 Entonces
          Escribir("No tiene altura suficiente Extranjero")
          Si edad < 18 Entonces
            Escribir("Tambien es menor de edad")
          FinSi
        SiNo
          Si edad < 18 Entonces
            Escribir("es menor de edad")
          FinSi
        FinSi
      FinSi
    SiNo
      Si edad < 18 Entonces
        Escribir("sale menor de edad")
      FinSi
      Si altura < 1.70 Entonces
        Escribir("sale altura insuficiente")
      FinSi
      Si extranjero = Falso Entonces
        Escribir("no es extranjero")
      FinSi
    FinSi
FinAccion
