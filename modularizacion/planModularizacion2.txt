  MEJORA #1 — ALTA PRIORIDAD
  Refactorizar CallableInvoker: separar resolución de invocación
================================================================================

PROBLEMA:
  CallableInvoker mezcla dos responsabilidades:
    1. Resolver si un nombre es builtin o user-defined (responsabilidad del Registry)
    2. Invocar la función/procedimiento encontrada (su responsabilidad real)
  Además, la lógica de "¿es callable?" está en CallableRegistry,
  pero la lógica de "¿cuál tipo es?" está en CallableInvoker — overlap confuso.

ARCHIVO ACTUAL AFECTADO:
  interpreter/callables/callableInvoker.ts

ACCIÓN: Crear archivo nuevo `interpreter/callables/callableResolver.ts`

CONTENIDO DE callableResolver.ts:
─────────────────────────────────
  export class CallableResolver {
    constructor(
      private readonly registry: CallableRegistry,
      private readonly builtinRegistry: BuiltinRegistry
    ) {}

    public resolveFunction(name: string): ResolvedCallable | RuntimeError {
      const builtin = this.builtinRegistry.resolve(name)
      if (builtin) return { kind: 'builtin', callable: builtin }

      const userFn = this.registry.getFunction(name)
      if (userFn) return { kind: 'function', declaration: userFn }

      if (this.registry.isProcedure(name)) {
        return new RuntimeError(FUNCTION_IS_PROCEDURE(name))
      }

      return new RuntimeError(NO_FUNCTION(name))
    }

    public resolveProcedure(name: string): ResolvedCallable | RuntimeError {
      const proc = this.registry.getProcedure(name)
      if (proc) return { kind: 'procedure', declaration: proc }

      if (this.registry.isFunction(name)) {
        return new RuntimeError(PROCEDURE_IS_FUNCTION(name))
      }

      return new RuntimeError(NO_PROCEDURE(name))
    }
  }

ACCIÓN: Refactorizar callableInvoker.ts para que SOLO invoque (no resuelva):
─────────────────────────────────────────────────────────────────────────────
  export class CallableInvoker {
    constructor(
      private readonly resolver: CallableResolver,
      private readonly executor: CallableExecutor
    ) {}

    public async invokeFunction(name: string, args: unknown[]): Promise<unknown> {
      const resolved = this.resolver.resolveFunction(name)
      if (resolved instanceof RuntimeError) throw resolved
      return this.executor.executeFunction(resolved.declaration, args)
    }

    public async invokeProcedure(name: string, args: unknown[]): Promise<void> {
      const resolved = this.resolver.resolveProcedure(name)
      if (resolved instanceof RuntimeError) throw resolved
      await this.executor.executeProcedure(resolved.declaration, args)
    }
  }

  MEJORA #2 — ALTA PRIORIDAD
  Implementar HandlerRegistry en StatementDispatcher (Open/Closed Principle)
================================================================================

PROBLEMA:
  StatementDispatcher.dispatch() contiene un switch/if-else con 10+ casos.
  Agregar un nuevo tipo de sentencia (ej: TryCatch, ForEach) requiere
  modificar directamente el dispatcher — viola Open/Closed Principle.

ARCHIVO ACTUAL AFECTADO:
  interpreter/orchestrator/statementDispatcher.ts

ACCIÓN: Crear archivo nuevo `interpreter/orchestrator/statementHandlerRegistry.ts`

CONTENIDO DE statementHandlerRegistry.ts:
──────────────────────────────────────────
  type StatementHandler = (
    node: StatementNode,
    context: EvaluatorContext
  ) => Promise<void>

  export class StatementHandlerRegistry {
    private readonly handlers = new Map<string, StatementHandler>()

    public register(type: string, handler: StatementHandler): void {
      this.handlers.set(type, handler)
    }

    public async dispatch(node: StatementNode, context: EvaluatorContext): Promise<void> {
      const handler = this.handlers.get(node.type)
      if (!handler) {
        throw new RuntimeError(`Tipo de sentencia no soportado: ${node.type}`)
      }
      await handler(node, context)
    }

    public supports(type: string): boolean {
      return this.handlers.has(type)
    }
  }

ACCIÓN: Refactorizar statementDispatcher.ts para usar el registry:
──────────────────────────────────────────────────────────────────
  export class StatementDispatcher {
    private readonly handlerRegistry: StatementHandlerRegistry

    constructor(
      private readonly ioEvaluator: IOEvaluator,
      private readonly controlFlowEvaluator: ControlFlowEvaluator,
      private readonly doWhileEvaluator: DoWhileEvaluator,
      private readonly switchEvaluator: SwitchEvaluator
    ) {
      this.handlerRegistry = new StatementHandlerRegistry()
      this.registerAllHandlers()
    }

    private registerAllHandlers(): void {
      this.handlerRegistry.register('Write', (node, ctx) =>
        this.ioEvaluator.evaluateWriteNode(node as WriteNode, ctx))
      this.handlerRegistry.register('Read', (node, ctx) =>
        this.ioEvaluator.evaluateReadNode(node as ReadNode, ctx))
      this.handlerRegistry.register('If', (node, ctx) =>
        this.controlFlowEvaluator.evaluateIfNode(node as IfNode, ctx))
      this.handlerRegistry.register('While', (node, ctx) =>
        this.controlFlowEvaluator.evaluateWhileNode(node as WhileNode, ctx))
      this.handlerRegistry.register('For', (node, ctx) =>
        this.controlFlowEvaluator.evaluateForNode(node as ForNode, ctx))
      this.handlerRegistry.register('Switch', (node, ctx) =>
        this.switchEvaluator.evaluateSwitchNode(node as SwitchNode, ctx))
      this.handlerRegistry.register('DoWhile', (node, ctx) =>
        this.doWhileEvaluator.evaluateDoWhileNode(node as DoWhileNode, ctx))
      // VariableDeclaration, Assignment, CallStatement se manejan inline
      // ya que acceden al environment directamente
    }

    public async dispatch(node: StatementNode, context: EvaluatorContext): Promise<void> {
      if (this.handlerRegistry.supports(node.type)) {
        return this.handlerRegistry.dispatch(node, context)
      }
      // Fallback para sentencias que no usan context complejo
      await this.dispatchSimple(node, context)
    }
  }

CRITERIO DE ÉXITO:
  - Agregar un nuevo tipo de sentencia = solo registrar un handler, sin tocar dispatch()
  - El switch/if-else original desaparece de dispatch()
  - StatementHandlerRegistry es exportado para uso externo si se necesita extensión

    debug(message: string, data?: unknown): void
    warn(message: string, data?: unknown): void
    error(message: string, error?: Error): void
  }

  export class ConsoleLogger implements InterpreterLogger {
    constructor(private readonly prefix = '[Interpreter]') {}

    debug(message: string, data?: unknown): void {
      if (process.env.NODE_ENV === 'development') {
        console.debug(`${this.prefix} ${message}`, data ?? '')
      }
    }

    warn(message: string, data?: unknown): void {
      console.warn(`${this.prefix} ${message}`, data ?? '')
    }

    error(message: string, error?: Error): void {
      console.error(`${this.prefix} ${message}`, error ?? '')
    }
  }

    debug(): void {}
    warn(): void {}
    error(): void {}
  }

────────────────────────────────────────────────────────────────
  export class LoopGuard {
    private static readonly DEFAULT_LIMIT = 10_000
    private iterations = 0

    constructor(private readonly limit: number = LoopGuard.DEFAULT_LIMIT) {}

    public checkIteration(loopType: string): void {
      this.iterations++
      if (this.iterations > this.limit) {
        throw new RuntimeError(INFINITE_LOOP_DETECTED(loopType, this.limit))
      }
    }

    public reset(): void {
      this.iterations = 0
    }

    public getIterations(): number {
      return this.iterations
    }

    // Solo para tests
    public static forTesting(limit: number = 100): LoopGuard {
      return new LoopGuard(limit)
    }
  }

      name: string,
      argNodes: ExpressionNode[],
